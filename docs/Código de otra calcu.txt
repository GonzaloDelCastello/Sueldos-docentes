function inicializarTabs() {
    const tabBtns = document.querySelectorAll(".tab-btn");
    const tabContents = document.querySelectorAll(".tab-content");

    tabBtns.forEach(btn => {
        btn.addEventListener("click", function () {
            tabBtns.forEach(b => b.classList.remove("activo"));
            tabContents.forEach(c => c.classList.remove("activo"));

            this.classList.add("activo");
            const target = this.getAttribute("data-tab");
            const targetEl = document.getElementById(target);
            if (targetEl) targetEl.classList.add("activo");
            window.scrollTo({ top: 0, behavior: "smooth" });
        });
    });

    // Activar pestaña según hash en la URL (ej: #tab-auditoria)
    if (window.location.hash) {
        const hashTab = window.location.hash.replace("#", "");
        const btnTab = document.querySelector(`.tab-btn[data-tab="${hashTab}"]`);
        if (btnTab) {
            btnTab.click();
        }
    }

    const linkAudit = document.getElementById("link-ir-auditoria");
    if (linkAudit) {
        linkAudit.addEventListener("click", function (e) {
            e.preventDefault();
            const btnAud = document.querySelector('[data-tab="tab-auditoria"]');
            if (btnAud) btnAud.click();
        });
    }
}

function inicializarEventosGlobales() {
    const btnAddCargo = document.getElementById("btn-add-cargo");
    if (btnAddCargo) {
        btnAddCargo.addEventListener("click", function () {
            agregarNuevoCargoDefault();
        });
    }

    const btnCalcular = document.getElementById("btn-calcular");
    if (btnCalcular) {
        btnCalcular.addEventListener("click", function () {
            calcularSueldoPluriempleo();
        });
    }

    const btnAuditAnses = document.getElementById("btn-calcular-auditoria");
    if (btnAuditAnses) {
        btnAuditAnses.addEventListener("click", ejecutarAuditoriaANSES);
    }

    const btnInf = document.getElementById("btn-comparar-inflacion");
    if (btnInf) {
        btnInf.addEventListener("click", ejecutarComparativaInflacion);
    }

    const btnPrint = document.getElementById("btn-imprimir-dictamen");
    if (btnPrint) {
        btnPrint.addEventListener("click", () => window.print());
    }
}

function agregarNuevoCargoDefault() {
    const cargoId = nextCargoId++;

    const nuevoCargo = {
        id: cargoId,
        establecimientoId: "", // Sin escuela precargada fija
        establecimientoNombre: "",
        filtroRegion: "todas",
        busquedaTexto: "",
        mostrarBuscador: true, // Inicia con el buscador abierto para que el docente elija o escriba su escuela
        nivel: "secundario",
        cargoTipo: "horaSecundaria",
        cantHoras: 15,
        zonaManual: "auto",
        antiguedadIdx: 5,
        presencialidad: true
    };

    listaCargosDocente.push(nuevoCargo);
    renderizarContenedorCargos();
}

function eliminarCargo(cargoId) {
    if (listaCargosDocente.length <= 1) {

    listaCargosDocente = listaCargosDocente.filter(c => c.id !== cargoId);
    renderizarContenedorCargos();
}

function obtenerColorPorZona(zona) {
    if (zona >= 100) return "#a855f7"; // Púrpura
    if (zona >= 80) return "#ec4899";  // Rosa intenso
    if (zona >= 60) return "#f97316";  // Naranja
    if (zona >= 40) return "#eab308";  // Ámbar
    if (zona === 10) return "#06b6d4"; // Cyan
    if (zona === 20) return "#3b82f6"; // Azul
    return "#64748b"; // Gris / Slate
}

function obtenerEscuelasFiltradasHTML(cargo) {
    const txt = (cargo.busquedaTexto || "").toLowerCase().trim();
    const rawTxt = (cargo.busquedaTexto || "").trim();
    const reg = cargo.filtroRegion || "todas";

    let filtradas = ESTABLECIMIENTOS_SL.filter(e => {
        if (e.id === "manual") return false;
        
        // Filtro por región
        if (reg !== "todas" && e.region !== reg) return false;

        // Filtro por texto
        if (txt) {
            const matchNombre = e.nombre.toLowerCase().includes(txt);
            const matchNumero = e.numero && e.numero.includes(txt);
            const matchLoc = e.localidad.toLowerCase().includes(txt);
            const matchDepto = (e.departamento || "").toLowerCase().includes(txt);
            return matchNombre || matchNumero || matchLoc || matchDepto;
        }
        return true;
    });

    let customOptionHtml = "";
    if (rawTxt.length >= 2) {
        const safeTxt = rawTxt.replace(/'/g, "\\'");
        customOptionHtml = `
            <div class="escuela-item-result escuela-custom-option" style="border-left: 4px solid #3b82f6; background: rgba(59, 130, 246, 0.1);" onclick="seleccionarEscuelaPersonalizada(${cargo.id}, '${safeTxt}')">
                <div class="escuela-item-main">
                    <span class="escuela-item-name">✍️ Usar "<strong>${rawTxt}</strong>" como nombre de mi escuela</span>
                    <span class="escuela-item-loc">📍 Establecimiento personalizado (Zona configurable)</span>
                </div>
                <div class="escuela-item-badge">
                    <span class="badge-zona-pill" style="background:#3b82f6;">Elegir Esta</span>
                </div>
            </div>
        `;
    }

    if (filtradas.length === 0) {
        return `
            ${customOptionHtml}
            <div class="escuela-no-results">
                <p>No se encontraron escuelas del catálogo oficial con "<strong>${cargo.busquedaTexto}</strong>".</p>
                <button type="button" class="btn-select-manual-fallback" onclick="seleccionarEscuelaPersonalizada(${cargo.id}, '${rawTxt.replace(/'/g, "\\'") || "Mi Escuela"}')">
                    📍 Establecer como "${rawTxt || "Escuela Personalizada"}"
                </button>
            </div>
        `;
    }

    return customOptionHtml + filtradas.map(e => `
        <div class="escuela-item-result ${e.id === cargo.establecimientoId ? 'seleccionada' : ''}" onclick="seleccionarEscuelaPorId(${cargo.id}, '${e.id}')">
            <div class="escuela-item-main">
                <span class="escuela-item-name">${e.nombre}</span>
                <span class="escuela-item-loc">📍 ${e.localidad} • ${e.departamento || "San Luis"} (${e.tipo})</span>
            </div>
            <div class="escuela-item-badge">
                <span class="badge-zona-pill" style="background:${obtenerColorPorZona(e.zona)};">
                    ${e.zona}% ZONA
                </span>
            </div>
        </div>
    `).join("");
}

function obtenerPorcentajeZonaCargo(cargo) {
    if (cargo.zonaManual !== null && cargo.zonaManual !== undefined && cargo.zonaManual !== "auto") {
        return parseInt(cargo.zonaManual);
    }
    if (cargo.establecimientoId && cargo.establecimientoId !== "manual" && cargo.establecimientoId !== "custom") {
        const est = ESTABLECIMIENTOS_SL.find(e => e.id === cargo.establecimientoId);
        return est ? est.zona : 0;
    }
    return 0;
}

window.actualizarBusquedaTexto = function(cargoId, txt) {
    const cargo = listaCargosDocente.find(c => c.id === cargoId);
    if (!cargo) return;
    cargo.busquedaTexto = txt;

    // Actualizar únicamente el contenedor de resultados de este puesto sin tocar el input para preservar foco y escritura continua
    const card = document.getElementById(`cargo-box-${cargoId}`);
    if (card) {
        const resDiv = card.querySelector(".lista-escuelas-resultados");
        if (resDiv) {
            resDiv.innerHTML = obtenerEscuelasFiltradasHTML(cargo);
        }
        const wrapper = card.querySelector(".input-search-wrapper");
        if (wrapper) {
            let clearBtn = wrapper.querySelector(".btn-clear-search");
            if (txt && !clearBtn) {
                clearBtn = document.createElement("button");
                clearBtn.type = "button";
                clearBtn.className = "btn-clear-search";
                clearBtn.innerHTML = "✕";
                clearBtn.onclick = () => window.limpiarBusquedaTexto(cargoId);
                wrapper.appendChild(clearBtn);
            } else if (!txt && clearBtn) {
                clearBtn.remove();
            }
        }
    }
};

window.limpiarBusquedaTexto = function(cargoId) {
    const cargo = listaCargosDocente.find(c => c.id === cargoId);
    if (!cargo) return;
    cargo.busquedaTexto = "";
    const card = document.getElementById(`cargo-box-${cargoId}`);
    if (card) {
        const input = card.querySelector(".input-search-escuela");
        if (input) {
            input.value = "";
            input.focus();
        }
        const resDiv = card.querySelector(".lista-escuelas-resultados");
        if (resDiv) {
            resDiv.innerHTML = obtenerEscuelasFiltradasHTML(cargo);
        }
        const clearBtn = card.querySelector(".btn-clear-search");
        if (clearBtn) clearBtn.remove();
    }
};

function calcularSueldoPluriempleo() {
    const periodoEl = document.getElementById("mesCalculo");
    const periodo = periodoEl ? periodoEl.value : "2026-08";
    
    const antEl = document.getElementById("antiguedad");
    const antIdx = antEl ? (parseInt(antEl.value) || 0) : 5;
    
    const sindEl = document.getElementById("sindicato");
    const sindicatoVal = sindEl ? (parseInt(sindEl.value) || 0) : 0;
    
    const hijosEl = document.getElementById("cantHijxs");
    const cantHijos = hijosEl ? (parseInt(hijosEl.value) || 0) : 0;
    const pctAnt = ESCALA_ANTIGUEDAD[antIdx] !== undefined ? ESCALA_ANTIGUEDAD[antIdx] : 0.50;

    let subtotalRemunGlobal = 0.0;
    let subtotalNoRemunGlobal = 0.0;
    let basicoGlobal = 0.0;
    let zonaGlobal = 0.0;
    let antGlobal = 0.0;
    let compRemGlobal = 0.0;
    let adicCargoGlobal = 0.0;
    let presencialidadGlobal = 0.0;
    let compNoRemGlobal = 0.0;
    let sumaNoRemGlobal = 0.0;
    let fonidGlobal = 0.0;
    let bonoExtGlobal = 0.0;

    const desglosePorPuesto = [];
    const desglosesCargos = [];

    listaCargosDocente.forEach((cargo, idx) => {
        const isIfdc = cargo.nivel === "superior" || (cargo.cargoTipo && cargo.cargoTipo.startsWith("ifdc"));
        const periodoClean = periodo.replace("-SAC", "");
        const listaConfigs = isIfdc ? HISTORIAL_IFDC : HISTORIAL_BASICA;
        let config = listaConfigs.find(c => c.fecha === periodoClean) || listaConfigs[listaConfigs.length - 1];

        const pctZona = obtenerPorcentajeZonaCargo(cargo);
        const coefZona = pctZona / 100.0;

        let basico = 0.0;
        let pagoZona = 0.0;
        let pagoAntiguedad = 0.0;
        let compRem = 0.0;
        let adicionalTurno = 0.0;
        let adicionalSeccion = 0.0;
        let adicionalCargo = 0.0;
        let adicionalDedicacion = 0.0;
        let enseñanzaEnAula = 0.0;
        let compNoRem = 0.0;
        let sumaNoRemun = 0.0;
        let fonid = 0.0;
        let compCompensador = 0.0;
        let bonoExt = 0.0;

        const cantHoras = Number(cargo.cantHoras) || 15;

        if (cargo.cargoTipo === "horaSecundaria") {
            const valorHora = config.basicoCargo_Hora || 18505.56;
            basico = valorHora * cantHoras;
            pagoZona = basico * coefZona;
            pagoAntiguedad = basico * pctAnt;
            compRem = basico * 1.40;
            compNoRem = basico * 0.97;
            sumaNoRemun = 4667.48 * cantHoras;
            fonid = 773.12 * cantHoras;
            if (cantHoras === 1) {
                compCompensador = 4018.89; // Código 100-26 Complemento Compensador para 1 hora
            }
        } else if (cargo.cargoTipo === "regente") {
            // Regente Est. Edu. Niv. Secundario 544 pts (Recibo Oficial)
            basico = 1170377.64;
            pagoAntiguedad = basico * pctAnt;
            pagoZona = basico * coefZona;
            adicionalTurno = basico * 0.13235294; // Código 211-10 Adicional por Turnos ($154,903.08)
            adicionalSeccion = basico * 0.29411765; // Código 212-10 Adicional por Secciones ($344,228.63)
            sumaNoRemun = 254493.10; // Código 099-41
            fonid = 48895.68; // Código 090-11
        } else if (cargo.cargoTipo === "directorSec") {
            basico = 1450000.00;
            pagoAntiguedad = basico * pctAnt;
            pagoZona = basico * coefZona;
            adicionalTurno = basico * 0.13235294;
            adicionalSeccion = basico * 0.29411765;
            sumaNoRemun = 300000.00;
            fonid = 48895.68;
        } else if (cargo.cargoTipo === "viceDirectorSec") {
            basico = 1350000.00;
            pagoAntiguedad = basico * pctAnt;

        } else if (cargo.cargoTipo && cargo.cargoTipo.startsWith("ifdc")) {
            // Nivel Superior / IFDC Art. 6º Ley XV-0555-2007
            if (cargo.cargoTipo === "ifdcFullTime") {
                basico = 647694.60;
                sumaNoRemun = 163361.92;
                fonid = 27059.20;
            } else if (cargo.cargoTipo === "ifdcTiempoCompleto" || cargo.cargoTipo === "ifdcExclusivo") {

                sumaNoRemun = 140024.50;
                fonid = 23193.60;

                basico = 462639.00;
                sumaNoRemun = 116687.08;
                fonid = 19328.00;
            } else if (cargo.cargoTipo === "ifdcSemiExclusivo20") {
                basico = 370111.20;
                sumaNoRemun = 93349.67;
                fonid = 15462.40;
            } else if (cargo.cargoTipo === "ifdcSimple12") {
                basico = 222066.72;
                sumaNoRemun = 56009.80;
                fonid = 9277.44;
            } else if (cargo.cargoTipo === "ifdcSimple6") {
                basico = 111033.36;
                sumaNoRemun = 28004.90;
                fonid = 4638.72;
            } else if (cargo.cargoTipo === "ifdcDocenteAuxiliar25") {
                basico = 388616.76;
                sumaNoRemun = 98017.15;
                fonid = 16235.52;
            } else if (cargo.cargoTipo === "ifdcDocenteAuxiliar15") {
                basico = 233170.06;
                sumaNoRemun = 58810.29;
                fonid = 9741.31;
            } else if (cargo.cargoTipo === "ifdcBibliotecarioAuxiliar") {

            } else if (cargo.cargoTipo === "ifdcHorasReloj") {
                basico = 18505.56 * cantHoras;
                sumaNoRemun = 4667.48 * cantHoras;
                fonid = 773.12 * cantHoras;
            } else {
                basico = config.basicoCargo_Hora * (COEFICIENTES_CARGOS[cargo.cargoTipo] || 1.0);
                sumaNoRemun = config.sumaNoRemunerativa;
                fonid = config.fonid;
            }

            pagoZona = basico * coefZona;
            pagoAntiguedad = basico * pctAnt;
            compRem = basico * 1.40;
            compNoRem = basico * 0.97;
        } else {
            // Cargos Estatutarios Primaria / Inicial / Preceptor
            if (cargo.cargoTipo === "maestroGrado") {
                basico = 277583.40;
                sumaNoRemun = 70012.25;
                fonid = 11596.80;
                enseñanzaEnAula = cargo.presencialidad ? 125000.0 : 0.0;

                sumaNoRemun = 71000.00;

            } else if (cargo.cargoTipo === "preceptor") {
                basico = 262000.00;
                sumaNoRemun = 65000.00;
                fonid = 11596.80;
            } else if (cargo.cargoTipo === "asesorPedagogico") {
                basico = 512000.00;
                sumaNoRemun = 130000.00;
                fonid = 23193.60;
            } else {
                const coef = COEFICIENTES_CARGOS[cargo.cargoTipo] || 15.0;
                basico = (config.basicoCargo_Hora || 18505.56) * coef;
                sumaNoRemun = 4667.48 * 15;
                fonid = 773.12 * 15;
            }

            pagoZona = basico * coefZona;
            pagoAntiguedad = basico * pctAnt;
            compRem = basico * 1.40;
            compNoRem = basico * 0.97;
        }

        const totalRemPuesto = basico + compRem + adicionalTurno + adicionalSeccion + adicionalCargo + adicionalDedicacion + pagoZona + pagoAntiguedad + enseñanzaEnAula;

        adicCargoGlobal += (adicionalTurno + adicionalSeccion + adicionalCargo + adicionalDedicacion);

        bonoExtGlobal += bonoExt;

        const est = cargo.establecimientoId && cargo.establecimientoId !== 'manual' && cargo.establecimientoId !== 'custom'

        });

        desglosesCargos.push({
            cargo,
            est,
            basico,
            pagoZona,
            pagoAntiguedad,
            compRem,
            adicionalTurno,
            adicionalSeccion,
            adicionalCargo,

    // Descuentos de Ley Docente (13% Jubilación Docente + 6% DOSEP)
    const descJubOrd = subtotalRemunGlobal * 0.11;
    const descJubEsp = subtotalRemunGlobal * 0.02;

    const sacBruto = subtotalRemunGlobal / 2.0;

    const sacNeto = sacBruto - descSacLey;

    let bolsilloGlobal = totalBrutoGlobal - totalDescuentos;
    if (periodo.endsWith("-SAC")) {
        bolsilloGlobal += sacNeto;
    }

    ultimoCalculoGlobal = {
        periodo,
        cantidadPuestos: listaCargosDocente.length,

function numeroALetrasJS(num) {
    if (isNaN(num)) return "";
    const entero = Math.floor(Math.abs(num));
    const centavos = Math.round((Math.abs(num) - entero) * 100);
    const centavosStr = centavos.toString().padStart(2, "0") + "/100";

    function unidades(n) {
        switch (n) {
            case 1: return "UN";
            case 2: return "DOS";
            case 3: return "TRES";
            case 4: return "CUATRO";
            case 5: return "CINCO";
            case 6: return "SEIS";
            case 7: return "SIETE";
            case 8: return "OCHO";
            case 9: return "NUEVE";
            default: return "";
        }
    }
    function decenasY(str, n) {
        if (n > 0) return str + " Y " + unidades(n);
        return str;
    }
    function decenas(n) {
        const d = Math.floor(n / 10);
        const u = n - (d * 10);
        switch (d) {
            case 1:
                switch (u) {
                    case 0: return "DIEZ";
                    case 1: return "ONCE";
                    case 2: return "DOCE";
                    case 3: return "TRECE";
                    case 4: return "CATORCE";
                    case 5: return "QUINCE";
                    default: return "DIECI" + unidades(u);
                }
            case 2:
                if (u === 0) return "VEINTE";
                return "VEINTI" + unidades(u);
            case 3: return decenasY("TREINTA", u);
            case 4: return decenasY("CUARENTA", u);
            case 5: return decenasY("CINCUENTA", u);
            case 6: return decenasY("SESENTA", u);
            case 7: return decenasY("SETENTA", u);
            case 8: return decenasY("OCHENTA", u);
            case 9: return decenasY("NOVENTA", u);
            case 0: return unidades(u);
            default: return "";
        }
    }
    function centenas(n) {
        const c = Math.floor(n / 100);
        const d = n - (c * 100);
        switch (c) {
            case 1:
                if (d > 0) return "CIENTO " + decenas(d);
                return "CIEN";
            case 2: return "DOSCIENTOS " + decenas(d);
            case 3: return "TRESCIENTOS " + decenas(d);
            case 4: return "CUATROCIENTOS " + decenas(d);
            case 5: return "QUINIENTOS " + decenas(d);
            case 6: return "SEISCIENTOS " + decenas(d);
            case 7: return "SETECIENTOS " + decenas(d);
            case 8: return "OCHOCIENTOS " + decenas(d);
            case 9: return "NOVECIENTOS " + decenas(d);
            default: return decenas(d);
        }
    }
    function seccion(n, divisor, strSingular, strPlural) {
        const cientos = Math.floor(n / divisor);
        const resto = n - (cientos * divisor);
        let letras = "";
        if (cientos > 0) {
            if (cientos > 1) letras = centenas(cientos) + " " + strPlural;
            else letras = strSingular;
        }
        return { letras, resto };
    }
    function miles(n) {
        const divisor = 1000;
        const cientos = Math.floor(n / divisor);
        const resto = n - (cientos * divisor);
        let strMiles = seccion(n, divisor, "MIL", "MIL").letras;
        let strCentenas = centenas(resto);
        if (strMiles === "") return strCentenas;
        return (strMiles + " " + strCentenas).trim();
    }
    function millones(n) {
        const divisor = 1000000;
        const cientos = Math.floor(n / divisor);
        const resto = n - (cientos * divisor);
        let strMillones = seccion(n, divisor, "UN MILLON", "MILLONES").letras;
        let strMiles = miles(resto);
        if (strMillones === "") return strMiles;
        return (strMillones + " " + strMiles).trim();
    }

    let letras = millones(entero);
    if (!letras || letras === "") letras = "CERO";
    return `${letras} PESOS CON ${centavosStr}`;
}

window.descargarReciboPDF = function () {
    const element = document.getElementById("recibo-scq-print-area");
    if (!element) {
        window.print();
        return;
    }

    const btn = document.getElementById("btn-download-pdf-scq");
    const oldTxt = btn ? btn.innerHTML : "📥 Descargar Recibo (PDF)";
    if (btn) btn.innerHTML = "⏳ Generando PDF...";

    const periodoEl = document.getElementById("mesCalculo");
    const perStr = periodoEl ? periodoEl.value : "Haberes";

    const opt = {
        margin: [4, 4, 4, 4],
        filename: `Recibo_SCQ_${perStr}_Docente.pdf`,
        image: { type: 'jpeg', quality: 0.98 },
        html2canvas: { scale: 2, useCORS: true, letterRendering: true },
        jsPDF: { unit: 'mm', format: 'a4', orientation: 'landscape' }
    };

    if (window.html2pdf) {
        window.html2pdf().set(opt).from(element).save().then(() => {
            if (btn) btn.innerHTML = oldTxt;
        }).catch(err => {
            console.error(err);
            window.print();
            if (btn) btn.innerHTML = oldTxt;
        });
    } else {
        window.print();
        if (btn) btn.innerHTML = oldTxt;
    }
};

// AUDITORÍA PREVISIONAL 82% MÓVIL (ANSES)
// ==========================================
function ejecutarAuditoriaANSES() {
    let resActivo = ultimoCalculoGlobal;
    if (!resActivo) {
        resActivo = calcularSueldoPluriempleo();
    }
    if (!resActivo) return;

    const inpAnses = document.getElementById("haber-anses-input");
    const haberAnses = inpAnses ? (parseFloat(inpAnses.value) || 0) : 0;
    if (haberAnses <= 0) {
        alert("Por favor ingrese el haber bruto mensual que actualmente percibe de ANSES.");
        if (inpAnses) inpAnses.focus();
        return;
    }

    const criterioEl = document.getElementById("criterio-base");
    const criterio = criterioEl ? criterioEl.value : "integral";

    const baseIntegral = resActivo.totalBruto;
    const baseRemun = resActivo.totalRemunerativo;

    const haber82Integral = baseIntegral * 0.82;
    const haber82Remun = baseRemun * 0.82;

    const haber82Aplicable = (criterio === "integral") ? haber82Integral : haber82Remun;
    const diferencia = haber82Aplicable - haberAnses;
    const brechaPct = haber82Aplicable > 0 ? ((diferencia / haber82Aplicable) * 100) : 0;
    const retroactivo24m = diferencia > 0 ? (diferencia * 24 + (diferencia / 2) * 4) : 0;

    // Mostrar sección
    const seccionAudit = document.getElementById("seccion-resultados-auditoria");
    if (seccionAudit) seccionAudit.style.display = "block";

    // Valores
    const elActivo = document.getElementById("audit-val-activo");
    if (elActivo) elActivo.textContent = aPesos(baseIntegral);

    const el82 = document.getElementById("audit-val-82");
    if (el82) el82.textContent = aPesos(haber82Aplicable);

    const elAnses = document.getElementById("audit-val-anses");
    if (elAnses) elAnses.textContent = aPesos(haberAnses);

    const elDif = document.getElementById("audit-val-dif");
    if (elDif) elDif.textContent = aPesos(diferencia);

    const elBrecha = document.getElementById("audit-val-brecha");
    if (elBrecha) elBrecha.textContent = brechaPct.toFixed(2) + "%";

    const elRetro = document.getElementById("audit-val-retroactivo");
    if (elRetro) elRetro.textContent = aPesos(retroactivo24m);

    // Comparativa 3 columnas
    const compAnses = document.getElementById("comp-anses-val");
    if (compAnses) compAnses.textContent = aPesos(haberAnses);

    const compRemun = document.getElementById("comp-remun-val");
    if (compRemun) compRemun.textContent = aPesos(haber82Remun);

    const compIntegral = document.getElementById("comp-integral-val");
    if (compIntegral) compIntegral.textContent = aPesos(haber82Integral);

    // Veredicto
    const cardVeredicto = document.getElementById("card-veredicto");
    const badgeVeredicto = document.getElementById("audit-badge-veredicto");
    const tituloVeredicto = document.getElementById("audit-titulo-veredicto");
    const descVeredicto = document.getElementById("audit-desc-veredicto");

    if (brechaPct >= 15.0) {
        if (cardVeredicto) cardVeredicto.className = "card-veredicto veredicto-confiscatorio";
        if (badgeVeredicto) badgeVeredicto.textContent = "CONFISCATORIEDAD GRAVE DETECTADA (> 15%)";
        if (tituloVeredicto) tituloVeredicto.textContent = `Pérdida Confiscatoria del ${brechaPct.toFixed(1)}% (Doctrina CSJN "Gemelli" / "Badaro")`;
        if (descVeredicto) descVeredicto.textContent = `ANSES le está liquidando un importe significativamente inferior al 82% móvil garantizado por la Ley 24.016. La merma supera el umbral del 15% de confiscatoriedad fijado por la Corte Suprema de Justicia de la Nación. Su caso es plenamente viable para reclamo administrativo y demanda judicial de reajuste.`;
    } else if (brechaPct > 0) {
        if (cardVeredicto) cardVeredicto.className = "card-veredicto veredicto-moderado";
        if (badgeVeredicto) badgeVeredicto.textContent = "DIFERENCIA SALARIAL DETECTADA";
        if (tituloVeredicto) tituloVeredicto.textContent = `Brecha del ${brechaPct.toFixed(1)}% a favor del Docente`;
        if (descVeredicto) descVeredicto.textContent = `Existe una diferencia mensual no liquidada entre lo abonado por ANSES y el 82% móvil del cargo en actividad. Corresponde intimar el reajuste y el pago retroactivo acumulado de los últimos 24 meses.`;
    } else {

        if (descVeredicto) descVeredicto.textContent = "No se detectaron diferencias salariales perjudiciales en el período auditado respecto al valor activo declarado.";
    }

    // Texto de Dictamen Jurídico
    const textoLegalEl = document.getElementById("audit-texto-legal");
    if (textoLegalEl) {
        const fechaHoy = new Date().toLocaleDateString("es-AR", { day: '2-digit', month: '2-digit', year: 'numeric' });
        textoLegalEl.innerHTML = `
            <p><strong>INFORME DE AUDITORÍA PREVISIONAL DOCENTE (Régimen Especial Ley 24.016)</strong></p>
            <p><strong>Fecha:</strong> ${fechaHoy} | <strong>Dictaminante:</strong> Estudio Jurídico Sebastián Careaga Quiroga & Asociados</p>

// COMPARATIVA PODER ADQUISITIVO VS INFLACIÓN
// ==========================================
function ejecutarComparativaInflacion() {
    const selInicio = document.getElementById("inf-inicio");
    const selFin = document.getElementById("inf-fin");
    const fIni = selInicio ? selInicio.value : "2023-06";
    const fFin = selFin ? selFin.value : "2026-06";

    const vIni = HISTORIAL_HORA[fIni] || 4899.23;
    const vFin = HISTORIAL_HORA[fFin] || 16963.44;

    const aumentoDocentePct = ((vFin - vIni) / vIni) * 100.0;
    const keyInf = `${fIni}_${fFin}`;
    const inflacionPct = INFLACION_ACUMULADA_APROX[keyInf] || (INFLACION_ACUMULADA_APROX["2023-06_2026-06"] || 420.5);

    const perdidaPoderPct = inflacionPct - aumentoDocentePct;

    const seccionInf = document.getElementById("seccion-res-inflacion");
    if (seccionInf) seccionInf.style.display = "block";

    const elTitulo = document.getElementById("inf-res-titulo");
    if (elTitulo) {
        elTitulo.textContent = perdidaPoderPct > 0 
            ? `Pérdida del Poder Adquisitivo: -${perdidaPoderPct.toFixed(1)}%`
            : `Aumento Real de Poder Adquisitivo: +${Math.abs(perdidaPoderPct).toFixed(1)}%`;
    }

    const elDetalle = document.getElementById("inf-res-detalle");
    if (elDetalle) {
        elDetalle.innerHTML = `
            <div style="font-size:14px; line-height:1.6; margin-top:10px;">
                <p><strong>Valor Hora ${fIni}:</strong> ${aPesos(vIni)} ➔ <strong>Valor Hora ${fFin}:</strong> ${aPesos(vFin)}</p>
                <p><strong>Variación Salarial Docente:</strong> +${aumentoDocentePct.toFixed(1)}%</p>
                <p><strong>Inflación Acumulada (IPC):</strong> +${inflacionPct.toFixed(1)}%</p>
                <hr style="border-color:#334155; margin:10px 0;">
                <p style="color:${perdidaPoderPct > 0 ? '#ef4444' : '#10b981'}; font-weight:bold; font-size:15px;">
                    ${perdidaPoderPct > 0 
                        ? `📉 El valor de la hora cátedra docente perdió un ${perdidaPoderPct.toFixed(1)}% frente a la inflación en este período.`
                        : `📈 El valor de la hora cátedra docente superó a la inflación en un ${Math.abs(perdidaPoderPct).toFixed(1)}%.`}
                </p>
            </div>
        `;
    }

    if (seccionInf) {
        seccionInf.scrollIntoView({ behavior: "smooth", block: "start" });
    }
